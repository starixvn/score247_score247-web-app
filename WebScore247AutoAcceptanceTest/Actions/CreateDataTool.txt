Get First Letter
    [Arguments]    ${input}    ${result}=
    @{words} =    Split String    ${input}    ${SPACE}
    : FOR    ${letter}    IN    @{words}
    \    @{string} =    Split String To Characters    ${letter}
    \    ${result}=    Catenate    SEPARATOR=    ${result}    ${string}[0]
    \    ${result}=    Convert To Uppercase    ${result}
    [Return]    ${result}
